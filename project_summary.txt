try:
    from paddleocr import PaddleOCR

    PaddleOCR = None

# ====================================================
# 🔒 全局互斥锁 (Reentrant Lock)
# ====================================================
_ADB_LOCK = threading.RLock()

# ==========================================
# 🧠 全局共享资源 (OCR)
# ==========================================
class GlobalOCR:
    _engine = None
    _lock = threading.Lock()

    @classmethod
    def _init_engine(cls):
        if PaddleOCR is None:
            return None
        # 👇【关键修改】去掉 show_log=False，防止新版报错
        return PaddleOCR(use_angle_cls=False, lang='ch')

    @classmethod
    def safe_ocr(cls, img_array):
        if cls._engine is None:
            with cls._lock:
                if cls._engine is None:
                    cls._engine = cls._init_engine()
        
        if cls._engine is None:
            return []

        with cls._lock:
            try:
                # 直接传图片
                result = cls._engine.ocr(img_array)
                return result
            except Exception as e:
                print(f"[GlobalOCR] 识别报错: {e}")
                return []

class DeviceOperator:
    def __init__(self, serial, instance_id, data_dir_root="data"):
        self.serial = serial
        self.instance_id = instance_id
        self.logger = logging.getLogger(f"{instance_id}.Device")
        
        self.base_dir = os.path.join(data_dir_root, instance_id)
        self.snapshot_dir = os.path.join(self.base_dir, "snapshots")
        os.makedirs(self.snapshot_dir, exist_ok=True)

        self._is_connected = False

    # ==========================================
    # 🛡️ 装饰器 (用于动作，返回 Bool)
    # ==========================================
    def _safe_wrap(func):
        @wraps(func)
        def wrapper(self, *args, **kwargs):
            action_name = func.__name__
            with _ADB_LOCK:
                self._ensure_context()
                try:
                    result = func(self, *args, **kwargs)
                    return True if result is None else result
                except TargetNotFoundError:
                    self.logger.warning(f"[{action_name}] 未找到目标")
                    self._save_snapshot(f"{action_name}_404")
                    return False
                except Exception as e:
                    self.logger.error(f"[{action_name}] 异常: {e}")
                    self._save_snapshot(f"{action_name}_error")
                    return False
        return wrapper

    # ==========================================
    # 🔌 系统连接
    # ==========================================
    def connect(self):
        with _ADB_LOCK:
            try:
                uri = f"Android:///{self.serial}?cap_method=JAVACAP&touch_method=ADBTOUCH"
                connect_device(uri)
                self._is_connected = True
                self.logger.info(f"连接成功: {self.serial}")
                return True
            except Exception as e:
                self.logger.error(f"连接失败: {e}")
                self._is_connected = False
                return False

    def _ensure_context(self):
        if self._is_connected:
            try:
                set_current(self.serial)
            except Exception:
                pass

    def _save_snapshot(self, tag):
        try:
            timestamp = datetime.now().strftime("%H%M%S")
            filename = f"{timestamp}_{tag}.png"
            filepath = os.path.join(self.snapshot_dir, filename)
            snapshot(filename=filepath)
            return filepath
        except:
            return None

    # ==========================================
    # 👆 动作执行 (保留装饰器)
    # ==========================================
    @_safe_wrap
    def click(self, target, times=1, offset=(0, 0)):
        for _ in range(times):
            touch(target)
            if times > 1:
                sleep(random.uniform(0.2, 0.5))

    @_safe_wrap
    def swipe_screen(self, direction="up", scale=0.6, area=None):
        if area:
            x1, y1, x2, y2 = map(int, area)
            w, h = x2 - x1, y2 - y1
            base_x, base_y = x1, y1
        else:
            w, h = G.DEVICE.get_current_resolution()
            base_x, base_y = 0, 0
        cx, cy = base_x + w * 0.5, base_y + h * 0.5
        delta_h, delta_w = h * scale * 0.5, w * scale * 0.5
        if direction == "up":    start, end = (cx, cy + delta_h), (cx, cy - delta_h)
        elif direction == "down": start, end = (cx, cy - delta_h), (cx, cy + delta_h)
        elif direction == "left": start, end = (cx + delta_w, cy), (cx - delta_w, cy)
        elif direction == "right": start, end = (cx - delta_w, cy), (cx + delta_w, cy)
        else: return False
        swipe(start, end)
        sleep(0.8)

    @_safe_wrap
    def swipe_until_found(self, target, direction="up", scale=0.5, area=None, max_retry=5):
        for i in range(max_retry):
            time.sleep(0.5)
            if exists(target): return True
            self.logger.info(f"寻找目标... 滑动 #{i+1}")
            self.swipe_screen(direction=direction, scale=scale, area=area)
        return False

    @_safe_wrap
    def pinch(self, mode="in"):
        from airtest.core.api import pinch as at_pinch
        at_pinch(mode=mode, center=None)
        sleep(1.5)

    @_safe_wrap
    def input_text(self, content, confirm=True):

        sleep(1)

    @_safe_wrap
    def press_back(self):
        keyevent("BACK")
        sleep(1)

    @_safe_wrap
    def app_restart(self, package_name):
        from airtest.core.android.android import Android
        dev = G.DEVICE
        if isinstance(dev, Android):
            dev.stop_app(package_name)
            sleep(1)
            dev.start_app(package_name)
            sleep(5)

    @_safe_wrap
    def exists(self, target):
        return True if exists(target) else False

    @_safe_wrap
    def sleep(self, seconds):
        sleep(seconds)

    # ==========================================
    # 👁️ 数据获取 (无装饰器，手动控制返回字符串)
    # ==========================================
    def get_text(self, area=None):
        """OCR 识别 (手动加锁，返回字符串)"""
        with _ADB_LOCK:
            try:
                self._ensure_context()
                temp_path = os.path.join(self.snapshot_dir, "ocr_tmp.png")
                snapshot(filename=temp_path)
                img = cv2.imread(temp_path)
                if img is None: return ""

                if area:
                    x1, y1, x2, y2 = map(int, area)
                    h, w = img.shape[:2]
                    x1, y1 = max(0, x1), max(0, y1)
                    x2, y2 = min(w, x2), min(h, y2)
                    img = img[y1:y2, x1:x2]

                # 图像增强
                try:
                    height, width = img.shape[:2]
                    scale = 3
                    zoomed = cv2.resize(img, (width*scale, height*scale), interpolation=cv2.INTER_CUBIC)
                    gray = cv2.cvtColor(zoomed, cv2.COLOR_BGR2GRAY)
                    _, binary = cv2.threshold(gray, 180, 255, cv2.THRESH_BINARY_INV)
                    final_img = cv2.copyMakeBorder(binary, 30, 30, 30, 30, cv2.BORDER_CONSTANT, value=(255, 255, 255))
                    final_img = cv2.cvtColor(final_img, cv2.COLOR_GRAY2BGR)
                except:
                    final_img = img
            except Exception as e:
                self.logger.error(f"OCR前置操作异常: {e}")
                return ""

        # OCR (自带锁，移出 ADB 锁范围)
        res = GlobalOCR.safe_ocr(final_img)
        if not res: return ""

        try:
            if isinstance(res[0], dict):
                texts = res[0].get('rec_texts', [])
                txt = "".join(texts)
            elif isinstance(res[0], list):
                txt = "".join([line[1][0] for line in res[0]])
            else:
                txt = ""
            return txt.strip()
        except:
            return ""

    def parse_number(self, text):
        if not text: return 0
        s = str(text).lower().replace(" ", "").replace(",", "")
        multiplier = 1
        if 'k' in s:
            multiplier = 1000
            s = s.replace('k', '')
        elif 'm' in s:
            multiplier = 1000000
            s = s.replace('m', '')
        try:
            val = float(s)
            return int(val * multiplier)
        except:
            return 0

    def check_resolution(self):
        try:
            from airtest.core.android.adb import ADB
            adb = ADB(self.serial)
            output = adb.shell("wm size")
            if not output: return False
            import re
            match = re.search(r"(\d+)x(\d+)", output)
            if match:
                w, h = int(match.group(1)), int(match.group(2))
                if (w == 720 and h == 1280) or (w == 1280 and h == 720): return True
            return False
        except:
            return False

# -*- coding: utf-8 -*-
import os
import json
import logging

class GlobalConfig:
    _instance = None
    
    DEFAULT_CONFIG = {
        "language": "zh_CN",
        "close_to_tray": False,   # 默认关闭：点击关闭按钮直接退出
        "auto_connect": True,    # 启动时自动连接
        "adb_path": "",          # 自定义 ADB 路径
        "use_internal_adb": True # 是否使用内置 ADB
    }

    def __new__(cls, *args, **kwargs):
        if not cls._instance:
            cls._instance = super(GlobalConfig, cls).__new__(cls, *args, **kwargs)
        return cls._instance

    def __init__(self, config_path="config.json"):
        if hasattr(self, "_initialized"):
            return
        self.config_path = config_path
        self.config = self.DEFAULT_CONFIG.copy()
        self.logger = logging.getLogger("GlobalConfig")
        self.load()
        self._initialized = True

    def load(self):
        if os.path.exists(self.config_path):
            try:
                with open(self.config_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    # 合并配置，确保新字段存在
                    for key, val in data.items():
                        self.config[key] = val
            except Exception as e:
                self.logger.error(f"加载全局配置失败: {e}")

    def save(self):
        try:
            with open(self.config_path, 'w', encoding='utf-8') as f:
                json.dump(self.config, f, indent=4, ensure_ascii=False)
        except Exception as e:
            self.logger.error(f"保存全局配置失败: {e}")

    def get(self, key, default=None):
        return self.config.get(key, default)

    def set(self, key, value):
        self.config[key] = value
        self.save()

# 全局单例
global_config = GlobalConfig()

# -*- coding: utf-8 -*-
import os
import json
import threading
import logging
from PySide6.QtCore import QObject, Signal

class InstanceDataStore(QObject):
    """
    单个实例的数据管家 (Per-Instance Data Manager)
    
    职责：
    1. 管理 config.json (用户配置：读多写少)
    2. 管理 statistics.json (游戏数据：写多读少)
    3. 维护 runtime memory (内存状态：不存盘)
    4. 作为 UI 和 后台脚本 的通讯枢纽 (Signals)
    """

    # ==================================================
    # 📡 信号定义 (Signals for UI)
    # ==================================================
    
    # 1. 配置变更信号: (key_path, new_value)
    # 场景：代码里重置了配置 -> UI 复选框自动更新
    sig_config_changed = Signal(str, object)

    # 2. 统计/资源数据变更信号: (key_path, new_value)
    # 场景：脚本 OCR 读到了新体力 -> UI 进度条刷新
    sig_stats_updated = Signal(str, object)

    # 3. 运行时状态变更信号: (key, new_value)
    # 场景：ADB 断开连接 -> UI 状态灯变红
    sig_runtime_changed = Signal(str, object)

    # 4. 日志信号: (level, message)
    # 场景：把后台的 print/log 转发到 UI 的文本框里
    sig_log_msg = Signal(str, str)

    def __init__(self, instance_id, data_root="data"):
        super().__init__()
        self.instance_id = instance_id
        
        # --- 路径配置 ---
        # 你的结构: data/deep_sl_002/
        self.base_dir = os.path.join(data_root, instance_id)
        self.config_path = os.path.join(self.base_dir, "config.json")
        self.stats_path = os.path.join(self.base_dir, "statistics.json")

        # --- 线程安全锁 ---
        # RLock (可重入锁) 允许同一个线程多次获得锁，防止死锁
        self._lock = threading.RLock()

        # --- 内存数据容器 ---
        self._config = {}   # 对应 config.json
        self._stats = {}    # 对应 statistics.json
        self._runtime = {   # 纯内存状态 (关机即焚)
            "is_running": False,       # 调度器运行状态
            "connect_status": "OFFLINE", # ADB 连接状态
            "current_task": "Idle"     # 当前正在执行的任务名
        }

        # --- 初始化 ---
        self._ensure_dirs()
        self.load_all()

    def _ensure_dirs(self):
        """确保专属文件夹及子文件夹结构完整"""
        # 1. 创建实例主目录 (data/deep_sl_002)
        if not os.path.exists(self.base_dir):
            os.makedirs(self.base_dir, exist_ok=True)
            
        # 2. 创建日志目录 (data/deep_sl_002/log)
        log_dir = os.path.join(self.base_dir, "log")
        if not os.path.exists(log_dir):
            os.makedirs(log_dir, exist_ok=True)

        # 3. 创建截图目录 (data/deep_sl_002/snapshots)
        snap_dir = os.path.join(self.base_dir, "snapshots")
        if not os.path.exists(snap_dir):
            os.makedirs(snap_dir, exist_ok=True)

    def load_all(self):
        """加载所有持久化数据"""
        # 1. 加载配置 (带默认值)
        default_config = {
            "instance_name": self.instance_id,
            "adb_device_id": "",
            "modules": {
                "general": {
                    "auto_restart": True,
                    "total_march_slots": 4,
                    "diana_level": 5 
                }
        }
        self._load_json(self.config_path, self._config, default_config)

        # 2. 加载统计 (默认为空)
        default_stats = {
            "stamina": 0,
            "resources": {
                "food": 0,
                "wood": 0,
                "stone": 0,
                "iron": 0
            }
        }
        self._load_json(self.stats_path, self._stats, default_stats)

    def _load_json(self, path, container, default):
        """通用的安全读取逻辑 (带简单的二级补全)"""
        with self._lock:
            if os.path.exists(path):
                try:
                    with open(path, 'r', encoding='utf-8') as f:
                        saved_data = json.load(f)
                        
                        # 1. 先把默认值深拷贝进来
                        # (这样能保证拥有最全的字段结构)
                        import copy
                        merged_data = copy.deepcopy(default)
                        
                        # 2. 递归更新 (这里只做两层简单的合并，够用了)
                        for key, val in saved_data.items():
                            if isinstance(val, dict) and key in merged_data and isinstance(merged_data[key], dict):
                                # 如果是字典（比如 modules），进去更新里面的字段
                                merged_data[key].update(val)
                            else:
                                # 如果是普通值，直接覆盖
                                merged_data[key] = val
                        
                        # 3. 更新到内存容器
                        container.clear()
                        container.update(merged_data)
                        
                except Exception as e:
                    self.push_log(f"文件 {os.path.basename(path)} 损坏/解析失败: {e}", "ERROR")
                    # 损坏了就用默认值
                    container.clear()
                    container.update(default)
                    self._save_json(path, container)
            else:
                # 文件不存在，初始化为默认值
                container.update(default)
                self._save_json(path, container)

    def _save_json(self, path, container):
        """通用的安全保存逻辑"""
        with self._lock:
            try:
                # 临时写入再重命名，防止写入一半断电导致文件损坏 (Atomic Write 思想)
                temp_path = path + ".tmp"
                with open(temp_path, 'w', encoding='utf-8') as f:
                    json.dump(container, f, indent=4, ensure_ascii=False)
                
                if os.path.exists(path):
                    os.remove(path)
                os.rename(temp_path, path)
            except Exception as e:
                self.push_log(f"保存 {os.path.basename(path)} 失败: {e}", "ERROR")

    def save_config(self):
        """手动触发保存配置"""
        self._save_json(self.config_path, self._config)

    def save_stats(self):
        """手动触发保存统计"""
        self._save_json(self.stats_path, self._stats)

    # ==================================================
    # 🔧 数据操作接口 (Getters & Setters)
    # ==================================================

    # --- 1. 用户配置 (Config) ---
    
    def get_cfg(self, key_path, default=None):
        """
        读取配置
        用法: get_cfg("modules.hunting.enable")
        """
        return self._get_value_by_path(self._config, key_path, default)

    def set_cfg(self, key_path, value):
        """
        修改配置 -> 立即保存 -> 通知 UI
        用法: set_cfg("modules.hunting.enable", True)
        """
        # 如果值确实改变了
        if self._set_value_by_path(self._config, key_path, value):
            self.save_config() # 持久化
            self.sig_config_changed.emit(key_path, value) # 通知 UI

    # --- 2. 游戏数据 (Statistics) ---
    
    def get_stat(self, key_path, default=0):
        """
        读取游戏统计
        用法: get_stat("resources.wood")
        """
        return self._get_value_by_path(self._stats, key_path, default)

    def set_stat(self, key_path, value):
        """
        修改游戏数据 -> 立即保存 -> 通知 UI
        用法: set_stat("stamina", 80)
        """
        if self._set_value_by_path(self._stats, key_path, value):
            self.save_stats() # 持久化
            self.sig_stats_updated.emit(key_path, value) # 通知 UI

    # --- 3. 运行时状态 (Runtime Memory) ---

    def get_runtime(self, key, default=None):
        """读取内存状态"""
        with self._lock:
            return self._runtime.get(key, default)

    def set_runtime(self, key, value):
        """
        修改运行时状态 -> 仅内存 -> 通知 UI
        用法: set_runtime("connect_status", "ONLINE")
        """
        with self._lock:
            if self._runtime.get(key) != value:
                self._runtime[key] = value
                self.sig_runtime_changed.emit(key, value) # 通知 UI

    # ==================================================
    # 🔨 辅助工具 (Helpers)
    # ==================================================

    def push_log(self, message, level="INFO"):
        """
        发送日志信号 (给 UI 显示用)
        并不负责写 log 文件 (写文件由 logging 模块在别处处理)
        """
        self.sig_log_msg.emit(level, message)

    def _get_value_by_path(self, root_dict, path, default):
        """
        [内部工具] 解析点号路径: 'a.b.c' -> dict['a']['b']['c']
        """
        keys = path.split('.')
        data = root_dict
        with self._lock:
            try:
                for k in keys:
                    data = data[k]
                return data
            except (KeyError, TypeError):
                return default

    def _set_value_by_path(self, root_dict, path, value):
        """
        [内部工具] 设置点号路径: 'a.b.c' = value
        返回 True 表示值发生了变化 (需要保存/刷新)
        返回 False 表示值没变 (节省 IO)
        """
        keys = path.split('.')
        data = root_dict
        with self._lock:
            try:
                # 递归找到倒数第二层
                for k in keys[:-1]:
                    # 如果中间层不存在，自动创建空字典
                    data = data.setdefault(k, {})
                
                last_key = keys[-1]
                old_value = data.get(last_key)
                
                if old_value != value:
                    data[last_key] = value
                    return True # 变了
                return False # 没变
            except Exception as e:
                print(f"设置键值失败 {path}: {e}")
                return False

                    if content:

            with open(self.registry_path, 'w', encoding='utf-8') as f:
                json.dump(self._registry, f, indent=4, ensure_ascii=False)
        except Exception as e:
            self.logger.error(f"保存注册表失败: {e}")

    def _init_all_instances(self):
        """启动时加载所有已注册账号"""
        task_classes = get_all_tasks()

        for instance_id, info in self._registry.items():
            self.logger.info(f"正在加载实例: {info.get('display_name')} ({instance_id})")
            
            # 1. 加载 DataStore
            store = InstanceDataStore(instance_id, self.data_root)
            self.data_stores[instance_id] = store
            
            # 2. 加载 Scheduler
            try:
                device = DeviceOperator(info['serial'], instance_id)
                if device.connect():

                else:
                    self.logger.warning(f"实例 {instance_id} 设备连接失败")
                sched = InstanceScheduler(instance_id, device, store, task_classes)
                self.schedulers[instance_id] = sched
            except Exception as e:

    # ===========================================

    # ===========================================
    def create_instance(self, display_name, serial):
        safe_name = re.sub(r'[^a-zA-Z0-9]', '_', display_name).lower()
        instance_id = safe_name
        counter = 1
        while instance_id in self._registry:
            instance_id = f"{safe_name}_{counter}"
            counter += 1

        try:
            # 1. 写入注册表
            self._registry[instance_id] = {
                "display_name": display_name,
                "serial": serial,
                "created_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            }
            self._save_registry()

            # 2. 创建 DataStore
            new_store = InstanceDataStore(instance_id, self.data_root)
            new_store.set_cfg("meta.serial", serial)
            new_store.set_cfg("meta.alias", display_name)

        except Exception as e:

            if instance_id in self._registry:
                del self._registry[instance_id]
            return False, str(e)

    """

    # 1. 读取配置，找到真正的 adb.exe

        print("⚠️ [System] 未找到指定 ADB，将使用系统默认")
        return

    print(f"✅ [System] 全局 ADB 路径已锁定: {_TARGET_ADB_PATH}")

    # 2. 【黑魔法】保存原版的构造函数
    _original_init = ADB.__init__

    # 3. 【黑魔法】定义一个新的构造函数
    def _patched_init(self, serialno=None, adb_path=None, server_addr=None, **kwargs):
        # 关键点：如果调用时没指定 path，或者指定的是默认的，我们强行替换成我们的

        # 调用原版逻辑，但传入了我们的 path
        _original_init(self, serialno=serialno, adb_path=adb_path, server_addr=server_addr, **kwargs)

    # 4. 【黑魔法】把 Airtest 的构造函数换成我们的

# -*- coding: utf-8 -*-
import logging
import os

class GuiLogHandler(logging.Handler):
    """
    自定义日志处理器：拦截日志并转发给 DataStore -> UI
    """
    def __init__(self, datastore):
        super().__init__()
        self.datastore = datastore
        self.instance_id = datastore.instance_id
        # 设置格式: 只取消息内容
        self.setFormatter(logging.Formatter("%(message)s"))

    def emit(self, record):
        """
        当有日志产生时触发
        """
        try:
            msg = self.format(record)
            level = record.levelname
            # 拼装 ID，防止 UI 串台
            full_msg = f"{self.instance_id}|||{msg}"
            self.datastore.push_log(full_msg, level)
        except Exception:
            self.handleError(record)

def setup_file_logging(instance_id, log_dir):
    """
    给指定实例挂载文件日志功能
    :return: 创建好的 FileHandler 对象 (方便后续移除)
    """
    logger = logging.getLogger(instance_id)
    logger.setLevel(logging.INFO)
    
    # 1. 检查是否已经有文件处理器了 (防止重复添加)
    for h in logger.handlers:
        if isinstance(h, logging.FileHandler):
            return h

    # 2. 确保目录存在
    if not os.path.exists(log_dir):
        os.makedirs(log_dir, exist_ok=True)

    # 3. 创建文件处理器
    log_path = os.path.join(log_dir, "ksas.log")
    # mode='a' 追加模式，encoding='utf-8' 防止中文乱码
    file_handler = logging.FileHandler(log_path, encoding='utf-8', mode='a')
    
    # 4. 设置格式 (文件里需要带时间戳，方便查案)
    formatter = logging.Formatter('%(asctime)s - [%(name)s] - %(levelname)s - %(message)s')
    file_handler.setFormatter(formatter)
    
    # 5. 挂载上去
    logger.addHandler(file_handler)
    
    return file_handler

import logging

import threading

import numpy as np

from datetime import datetime

# Airtest 核心 API

# OCR 库引入
try:
    from paddleocr import PaddleOCR
except ImportError:
    PaddleOCR = None

# ====================================================
# 🔒 全局互斥锁 (Reentrant Lock)
# ====================================================
_ADB_LOCK = threading.RLock()

# ==========================================
# 🧠 全局共享资源 (OCR)
# ==========================================
class GlobalOCR:
    _engine = None
    _lock = threading.Lock()

    @classmethod
    def _init_engine(cls):
        if PaddleOCR is None:
            return None
        # 👇【关键修改】去掉 show_log=False，防止新版报错
        return PaddleOCR(use_angle_cls=False, lang='ch')

    @classmethod
    def safe_ocr(cls, img_array):
        if cls._engine is None:
            with cls._lock:
                if cls._engine is None:
                    cls._engine = cls._init_engine()
        
        if cls._engine is None:
            return []

        with cls._lock:
            try:
                # 直接传图片
                result = cls._engine.ocr(img_array)
                return result
            except Exception as e:
                print(f"[GlobalOCR] 识别报错: {e}")
                return []

# ==========================================
# 🛠️ 实例操作器
# ==========================================
class DeviceOperator:
    def __init__(self, serial, instance_id, data_dir_root="data"):
        self.serial = serial
        self.instance_id = instance_id
        self.logger = logging.getLogger(f"{instance_id}.Device")
        
        self.base_dir = os.path.join(data_dir_root, instance_id)
        self.snapshot_dir = os.path.join(self.base_dir, "snapshots")
        os.makedirs(self.snapshot_dir, exist_ok=True)

        self._is_connected = False

    # ==========================================
    # 🛡️ 装饰器 (用于动作，返回 Bool)
    # ==========================================
    def _safe_wrap(func):
        @wraps(func)
        def wrapper(self, *args, **kwargs):
            action_name = func.__name__
            with _ADB_LOCK:
                self._ensure_context()
                try:
                    result = func(self, *args, **kwargs)
                    return True if result is None else result
                except TargetNotFoundError:
                    self.logger.warning(f"[{action_name}] 未找到目标")
                    self._save_snapshot(f"{action_name}_404")
                    return False
                except Exception as e:
                    self.logger.error(f"[{action_name}] 异常: {e}")
                    self._save_snapshot(f"{action_name}_error")
                    return False
        return wrapper

    # ==========================================
    # 🔌 系统连接
    # ==========================================
    def connect(self):
        with _ADB_LOCK:
            try:
                uri = f"Android:///{self.serial}?cap_method=JAVACAP&touch_method=ADBTOUCH"
                connect_device(uri)
                self._is_connected = True
                self.logger.info(f"连接成功: {self.serial}")
                return True
            except Exception as e:
                self.logger.error(f"连接失败: {e}")
                self._is_connected = False
                return False

    def _ensure_context(self):
        if self._is_connected:
            try:
                set_current(self.serial)
            except Exception:
                pass

    def _save_snapshot(self, tag):
        try:
            timestamp = datetime.now().strftime("%H%M%S")
            filename = f"{timestamp}_{tag}.png"
            filepath = os.path.join(self.snapshot_dir, filename)
            snapshot(filename=filepath)
            return filepath
        except:
            return None

    # ==========================================
    # 👆 动作执行 (保留装饰器)
    # ==========================================
    @_safe_wrap
    def click(self, target, times=1, offset=(0, 0)):
        for _ in range(times):
            touch(target)
            if times > 1:
                sleep(random.uniform(0.2, 0.5))

    @_safe_wrap
    def swipe_screen(self, direction="up", scale=0.6, area=None):
        if area:
            x1, y1, x2, y2 = map(int, area)
            w, h = x2 - x1, y2 - y1
            base_x, base_y = x1, y1
        else:
            w, h = G.DEVICE.get_current_resolution()
            base_x, base_y = 0, 0
        cx, cy = base_x + w * 0.5, base_y + h * 0.5
        delta_h, delta_w = h * scale * 0.5, w * scale * 0.5
        if direction == "up":    start, end = (cx, cy + delta_h), (cx, cy - delta_h)
        elif direction == "down": start, end = (cx, cy - delta_h), (cx, cy + delta_h)
        elif direction == "left": start, end = (cx + delta_w, cy), (cx - delta_w, cy)
        elif direction == "right": start, end = (cx - delta_w, cy), (cx + delta_w, cy)
        else: return False
        swipe(start, end)
        sleep(0.8)

    @_safe_wrap
    def swipe_until_found(self, target, direction="up", scale=0.5, area=None, max_retry=5):
        for i in range(max_retry):
            time.sleep(0.5)
            if exists(target): return True
            self.logger.info(f"寻找目标... 滑动 #{i+1}")
            self.swipe_screen(direction=direction, scale=scale, area=area)
        return False

    @_safe_wrap
    def pinch(self, mode="in"):
        from airtest.core.api import pinch as at_pinch
        at_pinch(mode=mode, center=None)
        sleep(1.5)

    @_safe_wrap

    @_safe_wrap
    def press_back(self):

        sleep(1)

    @_safe_wrap
    def app_restart(self, package_name):
        from airtest.core.android.android import Android
        dev = G.DEVICE
        if isinstance(dev, Android):
            dev.stop_app(package_name)
            sleep(1)
            dev.start_app(package_name)
            sleep(5)

    @_safe_wrap
    def exists(self, target):
        return True if exists(target) else False

    @_safe_wrap
    def sleep(self, seconds):
        sleep(seconds)

    # ==========================================
    # 👁️ 数据获取 (无装饰器，手动控制返回字符串)
    # ==========================================
    def get_text(self, area=None):
        """OCR 识别 (手动加锁，返回字符串)"""
        with _ADB_LOCK:
            try:
                self._ensure_context()
                temp_path = os.path.join(self.snapshot_dir, "ocr_tmp.png")
                snapshot(filename=temp_path)
                img = cv2.imread(temp_path)
                if img is None: return ""

                if area:
                    x1, y1, x2, y2 = map(int, area)
                    h, w = img.shape[:2]
                    x1, y1 = max(0, x1), max(0, y1)
                    x2, y2 = min(w, x2), min(h, y2)
                    img = img[y1:y2, x1:x2]

                # 图像增强
                try:
                    height, width = img.shape[:2]
                    scale = 3
                    zoomed = cv2.resize(img, (width*scale, height*scale), interpolation=cv2.INTER_CUBIC)
                    gray = cv2.cvtColor(zoomed, cv2.COLOR_BGR2GRAY)
                    _, binary = cv2.threshold(gray, 180, 255, cv2.THRESH_BINARY_INV)
                    final_img = cv2.copyMakeBorder(binary, 30, 30, 30, 30, cv2.BORDER_CONSTANT, value=(255, 255, 255))
                    final_img = cv2.cvtColor(final_img, cv2.COLOR_GRAY2BGR)
                except:
                    final_img = img
            except Exception as e:
                self.logger.error(f"OCR前置操作异常: {e}")
                return ""

        # OCR (自带锁，移出 ADB 锁范围)
        res = GlobalOCR.safe_ocr(final_img)
        if not res: return ""

        try:
            if isinstance(res[0], dict):
                texts = res[0].get('rec_texts', [])
                txt = "".join(texts)
            elif isinstance(res[0], list):
                txt = "".join([line[1][0] for line in res[0]])
            else:
                txt = ""
            return txt.strip()
        except:
            return ""

    def parse_number(self, text):
        if not text: return 0
        s = str(text).lower().replace(" ", "").replace(",", "")
        multiplier = 1
        if 'k' in s:
            multiplier = 1000
            s = s.replace('k', '')
        elif 'm' in s:
            multiplier = 1000000
            s = s.replace('m', '')
        try:
            val = float(s)
            return int(val * multiplier)
        except:
            return 0

# -*- coding: utf-8 -*-
import os
import json
import threading
import logging
from PySide6.QtCore import QObject, Signal

class InstanceDataStore(QObject):
    """
    单个实例的数据管家 (Per-Instance Data Manager)
    
    职责：
    1. 管理 config.json (用户配置：读多写少)
    2. 管理 statistics.json (游戏数据：写多读少)
    3. 维护 runtime memory (内存状态：不存盘)
    4. 作为 UI 和 后台脚本 的通讯枢纽 (Signals)
    """

    # ==================================================
    # 📡 信号定义 (Signals for UI)
    # ==================================================
    
    # 1. 配置变更信号: (key_path, new_value)
    # 场景：代码里重置了配置 -> UI 复选框自动更新
    sig_config_changed = Signal(str, object)

    # 2. 统计/资源数据变更信号: (key_path, new_value)
    # 场景：脚本 OCR 读到了新体力 -> UI 进度条刷新
    sig_stats_updated = Signal(str, object)

    # 3. 运行时状态变更信号: (key, new_value)
    # 场景：ADB 断开连接 -> UI 状态灯变红
    sig_runtime_changed = Signal(str, object)

    # 4. 日志信号: (level, message)
    # 场景：把后台的 print/log 转发到 UI 的文本框里
    sig_log_msg = Signal(str, str)

    def __init__(self, instance_id, data_root="data"):
        super().__init__()
        self.instance_id = instance_id
        
        # --- 路径配置 ---
        # 你的结构: data/deep_sl_002/
        self.base_dir = os.path.join(data_root, instance_id)
        self.config_path = os.path.join(self.base_dir, "config.json")
        self.stats_path = os.path.join(self.base_dir, "statistics.json")

        # --- 线程安全锁 ---
        # RLock (可重入锁) 允许同一个线程多次获得锁，防止死锁
        self._lock = threading.RLock()

        # --- 内存数据容器 ---
        self._config = {}   # 对应 config.json
        self._stats = {}    # 对应 statistics.json
        self._runtime = {   # 纯内存状态 (关机即焚)
            "is_running": False,       # 调度器运行状态
            "connect_status": "OFFLINE", # ADB 连接状态
            "current_task": "Idle"     # 当前正在执行的任务名
        }

        # --- 初始化 ---
        self._ensure_dirs()
        self.load_all()

    def _ensure_dirs(self):
        """确保专属文件夹及子文件夹结构完整"""
        # 1. 创建实例主目录 (data/deep_sl_002)
        if not os.path.exists(self.base_dir):
            os.makedirs(self.base_dir, exist_ok=True)
            
        # 2. 创建日志目录 (data/deep_sl_002/log)
        log_dir = os.path.join(self.base_dir, "log")
        if not os.path.exists(log_dir):
            os.makedirs(log_dir, exist_ok=True)

        # 3. 创建截图目录 (data/deep_sl_002/snapshots)
        snap_dir = os.path.join(self.base_dir, "snapshots")
        if not os.path.exists(snap_dir):
            os.makedirs(snap_dir, exist_ok=True)

    def load_all(self):
        """加载所有持久化数据"""
        # 1. 加载配置 (带默认值)
        default_config = {
            "instance_name": self.instance_id,
            "adb_device_id": "",
            "modules": {
                "general": {
                    "auto_restart": True,
                    "total_march_slots": 4,
                    "diana_level": 5 
                }
        }
        self._load_json(self.config_path, self._config, default_config)

        # 2. 加载统计 (默认为空)
        default_stats = {
            "stamina": 0,
            "resources": {
                "food": 0,
                "wood": 0,
                "stone": 0,
                "iron": 0
            }
        }
        self._load_json(self.stats_path, self._stats, default_stats)

    def _load_json(self, path, container, default):
        """通用的安全读取逻辑 (带简单的二级补全)"""
        with self._lock:
            if os.path.exists(path):
                try:
                    with open(path, 'r', encoding='utf-8') as f:
                        saved_data = json.load(f)
                        
                        # 1. 先把默认值深拷贝进来
                        # (这样能保证拥有最全的字段结构)
                        import copy
                        merged_data = copy.deepcopy(default)
                        
                        # 2. 递归更新 (这里只做两层简单的合并，够用了)
                        for key, val in saved_data.items():
                            if isinstance(val, dict) and key in merged_data and isinstance(merged_data[key], dict):
                                # 如果是字典（比如 modules），进去更新里面的字段
                                merged_data[key].update(val)
                            else:
                                # 如果是普通值，直接覆盖
                                merged_data[key] = val
                        
                        # 3. 更新到内存容器
                        container.clear()
                        container.update(merged_data)
                        
                except Exception as e:
                    self.push_log(f"文件 {os.path.basename(path)} 损坏/解析失败: {e}", "ERROR")
                    # 损坏了就用默认值
                    container.clear()
                    container.update(default)
                    self._save_json(path, container)
            else:
                # 文件不存在，初始化为默认值
                container.update(default)
                self._save_json(path, container)

    def _save_json(self, path, container):
        """通用的安全保存逻辑"""
        with self._lock:
            try:
                # 临时写入再重命名，防止写入一半断电导致文件损坏 (Atomic Write 思想)
                temp_path = path + ".tmp"
                with open(temp_path, 'w', encoding='utf-8') as f:
                    json.dump(container, f, indent=4, ensure_ascii=False)
                
                if os.path.exists(path):
                    os.remove(path)
                os.rename(temp_path, path)
            except Exception as e:
                self.push_log(f"保存 {os.path.basename(path)} 失败: {e}", "ERROR")

    def save_config(self):
        """手动触发保存配置"""
        self._save_json(self.config_path, self._config)

    def save_stats(self):
        """手动触发保存统计"""
        self._save_json(self.stats_path, self._stats)

    # ==================================================
    # 🔧 数据操作接口 (Getters & Setters)
    # ==================================================

    # --- 1. 用户配置 (Config) ---
    
    def get_cfg(self, key_path, default=None):
        """
        读取配置
        用法: get_cfg("modules.hunting.enable")
        """
        return self._get_value_by_path(self._config, key_path, default)

    def set_cfg(self, key_path, value):
        """
        修改配置 -> 立即保存 -> 通知 UI
        用法: set_cfg("modules.hunting.enable", True)
        """
        # 如果值确实改变了
        if self._set_value_by_path(self._config, key_path, value):
            self.save_config() # 持久化
            self.sig_config_changed.emit(key_path, value) # 通知 UI

    # --- 2. 游戏数据 (Statistics) ---
    
    def get_stat(self, key_path, default=0):
        """
        读取游戏统计
        用法: get_stat("resources.wood")
        """
        return self._get_value_by_path(self._stats, key_path, default)

    def set_stat(self, key_path, value):
        """
        修改游戏数据 -> 立即保存 -> 通知 UI
        用法: set_stat("stamina", 80)
        """
        if self._set_value_by_path(self._stats, key_path, value):
            self.save_stats() # 持久化
            self.sig_stats_updated.emit(key_path, value) # 通知 UI

    # --- 3. 运行时状态 (Runtime Memory) ---

    def get_runtime(self, key, default=None):
        """读取内存状态"""
        with self._lock:
            return self._runtime.get(key, default)

    def set_runtime(self, key, value):
        """
        修改运行时状态 -> 仅内存 -> 通知 UI
        用法: set_runtime("connect_status", "ONLINE")
        """
        with self._lock:
            if self._runtime.get(key) != value:
                self._runtime[key] = value
                self.sig_runtime_changed.emit(key, value) # 通知 UI

    # ==================================================
    # 🔨 辅助工具 (Helpers)
    # ==================================================

    def push_log(self, message, level="INFO"):
        """
        发送日志信号 (给 UI 显示用)
        并不负责写 log 文件 (写文件由 logging 模块在别处处理)
        """
        self.sig_log_msg.emit(level, message)

    def _get_value_by_path(self, root_dict, path, default):
        """
        [内部工具] 解析点号路径: 'a.b.c' -> dict['a']['b']['c']
        """
        keys = path.split('.')
        data = root_dict
        with self._lock:
            try:
                for k in keys:
                    data = data[k]
                return data
            except (KeyError, TypeError):
                return default

    def _set_value_by_path(self, root_dict, path, value):
        """
        [内部工具] 设置点号路径: 'a.b.c' = value
        返回 True 表示值发生了变化 (需要保存/刷新)
        返回 False 表示值没变 (节省 IO)
        """
        keys = path.split('.')
        data = root_dict
        with self._lock:
            try:
                # 递归找到倒数第二层
                for k in keys[:-1]:
                    # 如果中间层不存在，自动创建空字典
                    data = data.setdefault(k, {})
                
                last_key = keys[-1]
                old_value = data.get(last_key)
                
                if old_value != value:
                    data[last_key] = value
                    return True # 变了
                return False # 没变
            except Exception as e:
                print(f"设置键值失败 {path}: {e}")
                return False

    def create_instance(self, display_name, serial):
        safe_name = re.sub(r'[^a-zA-Z0-9]', '_', display_name).lower()
        instance_id = safe_name
        counter = 1
        while instance_id in self._registry:
            instance_id = f"{safe_name}_{counter}"
            counter += 1

        try:
            # 1. 写入注册表
            self._registry[instance_id] = {
                "display_name": display_name,
                "serial": serial,
                "created_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            }
            self._save_registry()

            # 2. 创建 DataStore
            new_store = InstanceDataStore(instance_id, self.data_root)
            new_store.set_cfg("meta.serial", serial)
            new_store.set_cfg("meta.alias", display_name)
            self.data_stores[instance_id] = new_store

            # 3. 创建 Scheduler
            device = DeviceOperator(serial, instance_id)
            if device.connect():

            sched = InstanceScheduler(instance_id, device, new_store, task_classes)

            self.sig_instance_list_changed.emit()
            return True, instance_id

        except Exception as e:
            self.logger.error(f"创建实例失败: {e}")
            if instance_id in self._registry:
                del self._registry[instance_id]
            return False, str(e)

# -*- coding: utf-8 -*-
import time
import logging
import threading
from datetime import datetime
from PySide6.QtCore import QThread, Signal

# 引入日志桥梁
from logging_setup import GuiLogHandler, setup_file_logging

class InstanceScheduler(QThread):
    """
    ALAS 风格的智能调度器 (Event-Driven Timeline Scheduler)
    
    特点：
    1. 基于时间轴 (Next Run) 调度，而非死循环轮询。
    2. 支持事件唤醒 (Config/NextRun 变更立即响应)。
    3. 自动过滤无关信号 (体力变化不吵醒线程)。
    4. 集成日志桥梁，将后台日志转发给 UI。
    """
    
    # 信号：向 UI 发送日志 (Level, Message)
    # 注意：有了 GuiLogHandler 后，这个信号主要由 Handler 触发，
    # 但保留在这里作为备用或直接调用接口。
    sig_log = Signal(str, str)

    def __init__(self, instance_id, device, data, task_classes):
        """
        :param task_classes: 任务类列表 [HuntingTask, GatheringTask...]
        """
        super().__init__()
        self.instance_id = instance_id
        self.device = device
        self.data = data
        
        # 1. 规范化 Logger 命名: "实例ID.模块"
        # 这样 logging_setup 里的过滤器就能按实例ID拦截日志
        self.logger = logging.getLogger(f"{instance_id}.Scheduler")
        
        # 2. 挂载日志桥梁 (Rainbow Bridge)
        # 监听属于该实例的所有日志
        self.instance_root_logger = logging.getLogger(instance_id)
        self.instance_root_logger.setLevel(logging.INFO)
        self.gui_handler = GuiLogHandler(data)
        self.instance_root_logger.addHandler(self.gui_handler)
        
        import os
        log_dir = os.path.join(data.base_dir, "log")
        
        # 2. 调用 logging_setup.py 里的函数，把它挂上去
        self.file_handler = setup_file_logging(instance_id, log_dir)

        self._is_running = False
        
        # 3. 核心组件：智能闹钟
        self._wake_event = threading.Event()
        
        # 4. 信号连接
        # 配置变了 -> 必须醒
        self.data.sig_config_changed.connect(self.on_config_changed)
        # 数据变了 (next_run 清空了) -> 必须醒
        self.data.sig_stats_updated.connect(self.on_stat_changed)

        # 5. 装载任务实例
        self.tasks = []
        for TaskCls in task_classes:
            try:
                task_obj = TaskCls(device, data)
                self.tasks.append(task_obj)
            except Exception as e:
                self.logger.error(f"任务 {TaskCls} 初始化失败: {e}")

    # ==========================================
    # 📡 信号处理 (过滤网)
    # ==========================================
    
    def on_config_changed(self, key, value):
        """配置变更：无条件唤醒"""
        if self._is_running:
            self.logger.debug(f"配置变更 [{key}]，唤醒调度器")
            self._wake_event.set()

    def on_stat_changed(self, key, value):
        """
        数据变更：【关键过滤】
        只有当变动的是 'next_run' 时才唤醒。
        """
        if self._is_running and "next_run" in key:
            self.logger.info(f"任务计划变更 [{key}]，立即唤醒调度器")
            self._wake_event.set()

    # ==========================================
    # 🎮 控制接口
    # ==========================================

    def start_scheduler(self):
        if not self.isRunning():
            self._is_running = True
            self.data.set_runtime("is_running", True)
            self._wake_event.clear()
            self.start()
            self.logger.info("调度器已启动")

    def stop_scheduler(self):
        if self.isRunning():
            self._is_running = False
            self.data.set_runtime("is_running", False)
            # 按下闹钟，让它从 wait() 中醒来并退出循环
            self._wake_event.set()
            self.logger.info("正在停止调度器...")
            self.wait() # 等待线程安全退出
            
            # 移除日志桥梁，防止内存泄漏或重复打印
            self.instance_root_logger.removeHandler(self.gui_handler)
            self.instance_root_logger.removeHandler(self.file_handler)
            self.logger.info("调度器已停止")

    # ==========================================
    # 🔄 核心循环 (The Brain)
    # ==========================================

    def run(self):
        self.logger.info("调度器线程已启动")
        
        # =========== 👇 新增：飞行前安检 (Pre-flight Check) 👇 ===========
        self.logger.info("正在检查设备环境...", "INFO")
        
        # 1. 确保连接 (防止连不上)
        if not self.device.connect():
            self.logger.error("❌ 无法连接设备！请检查模拟器是否启动。", "ERROR")
            self.data.set_runtime("is_running", False) # 告诉 UI 我停了
            return # 直接退出线程
            
        # 2. 检查分辨率 (防止点歪)
        if not self.device.check_resolution():
            self.logger.error("❌ 分辨率不匹配！脚本要求: 720x1280 (允许横屏)", "ERROR")
            self.logger.error("请修改模拟器分辨率设置并重启模拟器。", "ERROR")
            self.data.set_runtime("is_running", False) # 告诉 UI 我停了
            return # 直接退出线程
            
        self.logger.info("✅ 环境检查通过，开始调度任务。", "INFO")


        while self._is_running:
            self._wake_event.clear()
            
            now = datetime.now()
            ready_tasks = []
            next_wake_up_time = None 

            # --- 1. 扫描阶段 ---
            for task in self.tasks:
                if not task.enabled:
                    continue 
                
                if now >= task.next_run:
                    ready_tasks.append(task)
                else:
                    if next_wake_up_time is None or task.next_run < next_wake_up_time:
                        next_wake_up_time = task.next_run

            # --- 2. 决策阶段 ---
            if ready_tasks:
                # A. 有活干：按优先级排序
                ready_tasks.sort(key=lambda t: t.priority)
                winner = ready_tasks[0]
                
                self.logger.info(f"开始执行任务: {winner.name}")
                
                # 执行任务
                winner.execute() 
                
                time.sleep(1)
            
            else:
                # B. 没活干：计算休眠时间
                if next_wake_up_time:
                    delta = (next_wake_up_time - now).total_seconds()
                    wait_seconds = max(1, delta)
                    msg = f"休眠中 (下次任务 {int(wait_seconds)}s 后)"
                else:
                    wait_seconds = 3600 
                    msg = "无任务，待机中..."
                
                self.data.set_runtime("current_task", msg)
                self.logger.info(msg) # 打印休眠日志，证明没死机
                
                # --- 3. 智能休眠 ---
                self._wake_event.wait(wait_seconds)
                
                if self._is_running:
                    self.logger.debug("调度器被唤醒/自然醒，重新评估...")

# Final Architecture Blueprint: KSAS V3.4 (Independent Control Edition)

**Core Design Philosophy:**
1.  **Three-Layer Decoupling:** Complete separation of UI (Display) / Scheduler (Control) / Task (Execution).
2.  **Per-Instance Isolation:** Each account possesses independent data, configuration, logs, and scheduling threads to prevent interference.
3.  **Registration-Based Management:** Only devices explicitly "added" by the user are managed by the system, eliminating chaos from automatic scanning.
4.  **Independent Control:** Each instance controls its own start/stop status, rejecting the crude "Start All" logic.

---

## 0. 🚀 The Launcher

* **Role:** The program's **Shell** and **Updater**.
* **Implementation:** Independent executable (e.g., `launcher.exe`).
* **Responsibilities:**
    1.  **Environment Self-Check:** Ensures the runtime environment is normal.
    2.  **Auto-Update:** Checks for versions online; downloads and replaces core files if updates are available.
    3.  **Invoke Core:** Uses a subprocess to call `python KSAS.py` to launch the main program. Users do not run the main script directly.

---

## 1. 🏢 The Instance Manager

* **Role:** The device **Registry Office** and **Status Monitor**.
* **Implementation:** Global Singleton `InstanceManager`.
* **Core Logic:**
    1.  **Startup Load:**
        * Performs an ADB scan upon startup.
        * Reads `data/device_registry.json` to load all known devices (connected or not).
        * Initializes `DataStore` and `Scheduler` for each registered device; sets initial status to `OFFLINE`.
    2.  **Manual Scan:**
        * Executes `adb devices` only when the UI calls `scan_devices()`.
        * **Registered Devices:** Updates status to `ONLINE`; UI icons light up.
        * **Unregistered Devices:** Returns a "Candidate List", **does not automatically add them**, waits for user decision.
    3.  **Manual Registration (Add Instance):**
        * User selects a new device in the UI and confirms; calls `add_instance(serial, alias)`.
        * Creates data objects and writes to `device_registry.json`.
    4.  **Manual Removal:** Completely deletes the instance memory object and registry entry.

---

## 2. 💾 Data Center (Per-Instance DataStore)

* **Role:** The **State Repository** for a single instance.
* **Implementation:** Ordinary Class, one object per `device_id`.
* **Responsibilities:**
    1.  **Connection Status:** Maintains `connection_status` (ONLINE/OFFLINE).
    2.  **Running Status:** Maintains `running_status` (IDLE/RUNNING/ERROR).
    3.  **Config Management:** Stores task toggles and parameter settings for the instance (modifiable even when offline).
    4.  **Signal Driving:** Emits Qt Signals to notify the UI when status or configuration changes.

---

## 3. 🗄️ Persistence System (Structured Persistence)

* **Role:** File System Mapping.
* **Directory Structure:**
    ```text
    root/
    ├── data/
    │   ├── global_settings.json       # Global Settings (ADB path, webhook URL, UI prefs)
    │   ├── device_registry.json       # Device Registry (ID, Alias, Last Online Time)
    │   │
    │   ├── [device_id_A]/             # Exclusive folder for Instance A
    │   │   ├── config.json            # Task Configuration
    │   │   ├── runtime.log            # Runtime Logs
    │   │   └── snapshots/             # Error Screenshot Storage
    │   │
    │   └── [device_id_B]/             # Exclusive folder for Instance B
    │       ├── ...
    ```

---

## 4. ⏰ Central Scheduler (Per-Instance Scheduler)

* **Role:** The **Independent Commander** for a single instance.
* **Implementation:** Inherits `QThread`, one thread per instance.
* **Control Logic:**
    1.  **Independent Start/Stop:** Provides `start_scheduler()` and `stop_scheduler()` methods for independent UI buttons.
    2.  **Safe Execution:**
        * Refuses to start if status is `OFFLINE`.
        * If user clicks "Stop", exits safely after the current task completes its atomic operation.
    3.  **Crash Prevention:** Captures all Task exceptions within `run()`, triggers a snapshot, records logs, and ensures the guardian thread does not crash.

---

## 5. 🛠️ Task Executor (Instance-Aware Tasks)

* **Role:** Business Logic Unit.
* **Implementation:** Subclass of `BaseTask`.
* **Responsibilities:**
    1.  **Context Awareness:** Must hold `instance_id` to read/write correct data.
    2.  **Logic Execution:** Read Config -> Call Toolbox -> Update Status.

---

## 6. 🧰 Toolbox (Toolbox & Snapshot)

* **Role:** Low-level Operations Library & Evidence Retention.
* **Implementation:** `basic.py` (Airtest Wrapper) + `snapshot_manager.py`.
* **Responsibilities:**
    1.  **Safe Invocation:** Encapsulates Airtest APIs, catches low-level exceptions, returns Boolean results.
    2.  **Snapshot Mechanism:**
        * Calls `adb screencap` when a task errors or a key step fails.
        * Saves the image to `data/[device_id]/snapshots/`.
        * Returns the image path to the logging system.

---

## 7. 🖥️ Native UI (Multi-Instance View)

* **Role:** Console / Dashboard.
* **Core Layout:**
    1.  **Top Toolbar:**
        * `[ 🔄 Refresh Connection ]`: Manually triggers ADB scan, updates online status.
        * `[ ➕ Add Device ]`: Pops up a dialog to select and add from scanned unknown devices.
        * `[ ⚙️ Global Settings ]`: Modifies general configs like ADB path.
    2.  **Instance Area (Tab Widget or List):**
        * **Dynamic Rendering:** Generates tabs based on `device_registry`.
        * **Independent Control Panel (Inside each tab):**
            * **Status Bar:** Displays Alias, Connection Status (Lit/Grayed), Current Stamina/AP.
            * **Operation Area:** `[ ▶ Start Script ]` / `[ ⏹ Stop Script ]` (Mutually exclusive buttons).
            * **Config Area:** Task checkbox list.
            * **Log Area:** Scrolling log exclusive to this instance (includes screenshot links).

---

## 8. 🔌 Core Glue (Glue Code)

* **Role:** Signal Routing.
* **Mechanism:**
    * **UI Button -> Scheduler:** Clicking "Start" for Instance A directly calls `instance_manager.get_scheduler(A).start()`.
    * **DataStore -> UI:** If only Instance A's stamina changes, only Instance A's UI controls are refreshed, avoiding a full interface redraw.

---

## 9. 📇 Task Registry (Task Registry)

* **Role:** Static Dictionary.
* **Responsibilities:** Maintains the mapping between Task Names and Classes for UI configuration list generation.

---

## 10. 📝 Log System (Instance-Isolated Logging)

* **Role:** Distributing Recorder.
* **Strategy:**
    1.  **Physical Isolation:** Logs are strictly written to `data/[device_id]/runtime.log`.
    2.  **Real-time Feedback:** Pushes log text (and screenshot paths) to the UI log window via Signals.

def get_text(key):
    """
    获取翻译文本
    用法: get_text("btn_start") -> "▶ 启动"
    """
    data = TEXTS.get(key)
    if not data:
        # 如果 Key 没找到，直接返回 Key 本身，方便调试发现漏了啥
        return f"[{key}]"
    
    # 优先取当前语言，取不到就降级用中文，还不行就返回空串
    return data.get(CURRENT_LANG, data.get("cn", ""))

def set_language(lang_code):
    """
    切换全局语言
    :param lang_code: 'cn' or 'en'
    """
    global CURRENT_LANG
    if lang_code in ["cn", "en"]:
        CURRENT_LANG = lang_code

# tasks/base_task.py
# -*- coding: utf-8 -*-

import os
import logging
from airtest.core.api import Template
from datetime import datetime, timedelta

class BaseTask:
    """
    所有任务的基类 (ALAS 风格)
    
    核心属性：
    1. name: 任务名 (用于在 JSON 里查找配置)
    2. priority: 优先级 (数字越大越重要，或者越小越重要，这里我们定：数字越小优先级越高 1 > 10)
    3. default_interval: 默认间隔秒数 (如果 config 没填就用这个)
    """
    DIANA_COST_TABLE = {
        1: (23, 9),
        2: (22, 9),
        3: (22, 9),
        4: (21, 9),
        5: (20, 8)
    }

    # 默认消耗 (没有戴安娜时)
    DEFAULT_COST = (25, 10)



    def __init__(self, device, data):
        """
        :param device: basic.py 的 DeviceOperator
        :param data: datastore.py 的 InstanceDataStore
        """
        self.device = device
        self.data = data
        self.logger = logging.getLogger(f"{self.data.instance_id}.Task.{self.name}")

    # ==========================================
    # ⚙️ 子类必须重写/配置的属性
    # ==========================================
    @property
    def name(self):
        return "BaseTask"

    @property
    def priority(self):
        """优先级：1 (最高/紧急) ~ 100 (低/闲时)"""
        return 50

    @property
    def default_interval(self):
        """默认循环间隔 (秒)，默认 1 小时"""
        return 3600

    # ==========================================
    # ⏱️ ALAS 核心调度逻辑 (Scheduler 会调用这些)
    # ==========================================
    
    @property
    def enabled(self):
        """检查用户是否在 UI 上开启了这个任务"""
        # 读取 config.json: modules.hunting.enable
        return self.data.get_cfg(f"modules.{self.name.lower()}.enable", False)

    @property
    def next_run(self):
        """
        从 DataStore 读取下次运行时间
        :return: datetime 对象
        """
        # 读取 statistics.json: tasks.Hunting.next_run
        # 格式是字符串 "2023-11-19 12:00:00"
        time_str = self.data.get_stat(f"tasks.{self.name}.next_run", "1970-01-01 00:00:00")
        try:
            return datetime.strptime(time_str, "%Y-%m-%d %H:%M:%S")
        except ValueError:
            return datetime.min # 如果格式错，立即执行

    def set_next_run(self, seconds=None):
        """
        设置下次运行时间
        :param seconds: 多少秒后运行。如果不填，尝试去 config 读取 'interval'
        """
        if seconds is None:
            # 尝试从配置读取用户设定的间隔，没有则用默认
            seconds = self.data.get_cfg(f"modules.{self.name.lower()}.interval", self.default_interval)
        
        future_time = datetime.now() + timedelta(seconds=int(seconds))
        time_str = future_time.strftime("%Y-%m-%d %H:%M:%S")
        
        # 保存进 statistics.json (这就是日程表)
        self.data.set_stat(f"tasks.{self.name}.next_run", time_str)
        self.logger.info(f"下次运行安排在: {time_str}")

    # ==========================================
    # 🏃 运行入口 (不要重写这个，重写 run_impl)
    # ==========================================
    def execute(self):
        """
        调度器调用的标准入口
        1. 检查是否真的要跑
        2. 执行逻辑
        3. 自动更新下次时间
        """
        self.logger.info(f"=== 任务启动: {self.name} ===")
        
        # 1. 更新 UI 状态
        self.data.set_runtime("current_task", self.name)
        
        # 2. 执行具体逻辑 (子类实现)
        try:
            success = self.run_impl()
        except Exception as e:
            self.logger.error(f"任务崩溃: {e}")
            success = False
        
        # 3. 完事后，安排下次时间
        # 如果任务失败了，是立即重试还是按原计划？这里暂定按原计划
        # ALAS 的做法是：如果失败，可能只 sleep 5分钟而不是 1小时
        if success:
            self.set_next_run() # 用 config 里的正常间隔
        else:
            self.set_next_run(seconds=300) # 失败后 5 分钟重试
            
        self.logger.info(f"=== 任务结束: {self.name} ===")

    def run_impl(self):
        """
        【子类请重写这个方法】
        真正的业务逻辑写在这里 (打怪、采集等)
        :return: True(成功) / False(失败)
        """
        raise NotImplementedError

    # ==========================================
    # 🛠️ 常用快捷指令 (Shortcut)
    # ==========================================
    def screenshot(self):
        """任务中截图"""
        # 调用 device 的截图功能，tag 是任务名
        pass # 实际调用 basic.py

    def calculate_cost(self, mode="rally"):
        """
        计算本次出征的体力消耗
        :param mode: 'rally' (集结) 或 'beast' (打野)
        :return: int (消耗值)
        """
        # 1. 先看戴安娜在不在
        if self.find_diana():
            # 2. 如果在，去读配置里的技能等级
            # 我们把这个配置放在 general 里，因为它是账号属性
            skill_level = self.data.get_cfg("modules.general.diana_level", 1)
            
            # 查表
            costs = self.DIANA_COST_TABLE.get(skill_level, self.DEFAULT_COST)
            
            cost = costs[0] if mode == "rally" else costs[1]
            self.logger.info(f"触发戴安娜减耗 (Lv.{skill_level}): 本次消耗 {cost}")
            return cost
        
        else:
            # 3. 不在，用默认值
            cost = self.DEFAULT_COST[0] if mode == "rally" else self.DEFAULT_COST[1]
            self.logger.info(f"常规消耗: {cost}")
            return cost


    # ==========================================
    # 📊 游戏信息读取 (Game Info)
    # ==========================================

    def find_diana(self):
        """
        在出征界面寻找 Diana 的头像
        """
        # 记得把 Diana 的头像截图放在 assets/images/heroes/diana.png
        # 建议截那个圆圆的头像，特征明显一点
        img_diana = Template(r"assets/images/heroes/diana.png", threshold=0.9)
        
        if self.device.exists(img_diana):
            self.logger.info("检测到队伍包含 Diana")
            return True
        else:
            self.logger.debug("队伍中未检测到 Diana")
            return False

from PySide6.QtWidgets import (
    QDialog, QVBoxLayout, QHBoxLayout, QLabel, 
    QLineEdit, QComboBox, QPushButton, QMessageBox, QFormLayout
)
from airtest.core.android.adb import ADB
import os

class AddInstanceDialog(QDialog):
    def __init__(self, parent=None):
        super().__init__(parent)
        self.setWindowTitle("添加新实例")
        self.resize(400, 200)
        self.setup_ui()
        self.refresh_devices()

    def setup_ui(self):
        layout = QVBoxLayout(self)

        form = QFormLayout()
        
        # 实例名称
        self.line_name = QLineEdit()
        self.line_name.setPlaceholderText("例如: 主号, 小号1...")
        form.addRow("实例名称:", self.line_name)

        # 设备序列号
        hbox_serial = QHBoxLayout()
        self.combo_serial = QComboBox()
        self.combo_serial.setEditable(True)
        self.combo_serial.setPlaceholderText("选择或输入设备 Serial/IP:Port")
        
        btn_refresh = QPushButton("🔄")
        btn_refresh.setToolTip("刷新设备列表")
        btn_refresh.setFixedWidth(40)
        btn_refresh.clicked.connect(self.refresh_devices)
        
        hbox_serial.addWidget(self.combo_serial)
        hbox_serial.addWidget(btn_refresh)
        form.addRow("设备 Serial:", hbox_serial)

        layout.addLayout(form)

        # 按钮
        btn_layout = QHBoxLayout()
        btn_ok = QPushButton("确定")
        btn_ok.clicked.connect(self.accept)
        btn_cancel = QPushButton("取消")
        btn_cancel.clicked.connect(self.reject)
        
        btn_layout.addStretch()
        btn_layout.addWidget(btn_ok)
        btn_layout.addWidget(btn_cancel)
        layout.addLayout(btn_layout)

    def refresh_devices(self):
        self.combo_serial.clear()
        try:
            # 1. 获取配置的 ADB 路径
            from config import global_config
            adb_path = global_config.get("adb_path", "")
            use_internal = global_config.get("use_internal_adb", True)
            
            # 如果使用内置，尝试构建路径 (这里假设在当前目录下有 adb/adb.exe)
            if use_internal:
                # 简单的内置路径推断，实际可能需要更严谨的判断
                if os.path.exists("adb/adb.exe"):
                    adb_path = os.path.abspath("adb/adb.exe")
                else:
                    # 如果没找到内置的，就让 Airtest 自己去找
                    adb_path = None
            
            # 2. 初始化 ADB
            if adb_path and os.path.exists(adb_path):
                adb = ADB(adb_path=adb_path)
            else:
                adb = ADB() # 尝试自动查找

            # 3. 获取设备列表
            devices = adb.devices() # returns list of (serial, status)
            
            count = 0
            for serial, status in devices:
                if status == 'device':
                    self.combo_serial.addItem(serial)
                    count += 1
            
            if count == 0:
                self.combo_serial.setPlaceholderText("未检测到在线设备")
            else:
                self.combo_serial.setPlaceholderText("请选择设备...")
                self.combo_serial.showPopup() # 自动弹出列表方便选择

        except Exception as e:
            print(f"Error refreshing devices: {e}")
            QMessageBox.warning(self, "ADB 错误", f"无法获取设备列表:\n{e}")

    def get_data(self):
        return self.line_name.text().strip(), self.combo_serial.currentText().strip()

    def load_ui_file(self):
        loader = QUiLoader()
        ui_path = os.path.join(os.path.dirname(__file__), "main_window.ui")
        ui_file = QFile(ui_path)
        if not ui_file.open(QIODevice.ReadOnly):
            print(f"Cannot open {ui_path}: {ui_file.errorString()}")

        ui = loader.load(ui_file)
        ui_file.close()
        if not ui:
            print(loader.errorString())
            sys.exit(-1)
        return ui

            if global_config.get("close_to_tray", False):

                # 正常关闭退出

    def update_task_lists(self, sched):
        """根据调度器状态刷新中间的三个列表"""
        self.ui.list_running.clear()
        self.ui.list_queue.clear()
        self.ui.list_waiting.clear()
        
        if not sched: return

        store = self.manager.get_store(self.current_instance_id)
        
        # 获取当前正在运行的任务名
        current_task_name = store.get_runtime("current_task", "Idle")
        is_running_logic = store.get_runtime("is_running", False)

        # --- 填充 "运行中" ---
        if is_running_logic and "休眠" not in current_task_name and "待机" not in current_task_name and "Idle" not in current_task_name:
             self.ui.list_running.addItem(f"🔥 {current_task_name}")
        elif is_running_logic:
             self.ui.list_running.addItem(f"💤 {current_task_name}")
        
        # --- 准备数据 ---
        now = datetime.now()
        queue_tasks = []   # 时间到了
        waiting_tasks = [] # 时间没到

        for task in sched.tasks:
            if not task.enabled:
                continue
            
            # 格式化时间
            time_str = task.next_run.strftime("%H:%M:%S") if task.next_run.year > 2000 else "立即"
            
            # 已经在运行的就不显示在队列里了
            if task.name == current_task_name and "休眠" not in current_task_name:
                continue

            if now >= task.next_run:
                queue_tasks.append((task, time_str))
            else:
                waiting_tasks.append((task, time_str))

        # --- 填充 "队列中" (按优先级排序) ---
        # priority 越小越优先
        queue_tasks.sort(key=lambda x: x[0].priority)
        for task, t_str in queue_tasks:
            item = QListWidgetItem(f"[{task.priority}] {task.name}")
            item.setData(Qt.UserRole, task.name) # 存任务名方便右键操作
            self.ui.list_queue.addItem(item)

        # --- 填充 "等待中" (按时间排序) ---
        waiting_tasks.sort(key=lambda x: x[0].next_run)
        for task, t_str in waiting_tasks:
            item = QListWidgetItem(f"{task.name}  ({t_str})")
            item.setData(Qt.UserRole, task.name)
            self.ui.list_waiting.addItem(item)

    def reset_task_next_run(self, task_name):
        """重置任务时间 -> 立即执行"""

        """弹出对话框修改 Name 和 Serial"""
        
        # 1. 获取旧数据，用来填在框里
        store = self.manager.get_store(instance_id)
        # 从注册表里拿现在的名字（别名）
        all_insts = self.manager.get_all_instances()
        target = next((x for x in all_insts if x['id'] == instance_id), None)
        
        old_name = target['name'] if target else instance_id
        old_serial = store.get_cfg("meta.serial", "")
        
        # 2. 弹出对话框
        dialog = EditInstanceDialog(self.ui, current_name=old_name, current_serial=old_serial)
        
        if dialog.exec():
            new_name, new_serial = dialog.get_data()
            
            # 简单校验
            if not new_name or not new_serial:

                return

            # 3. 调用 Manager 更新
            # (注意：instance_manager.py 里的 update_instance 方法要确保已存在)
            self.manager.update_instance(instance_id, new_serial=new_serial, new_alias=new_name)

        name = target['name'] if target else instance_id

    def load_settings(self):

        # Behavior

        global_config.set("auto_connect", self.chk_auto_connect.isChecked())

            subprocess.run(["adb", "start-server"], check=False)
            QMessageBox.information(self, "成功", "ADB 服务已重启")
        except Exception as e:
            QMessageBox.warning(self, "失败", f"重启 ADB 失败: {e}")

    def backup_data(self):
        try:
            # 确保 backup 文件夹存在
            backup_dir = "backup"
            if not os.path.exists(backup_dir):
                os.makedirs(backup_dir)

            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = os.path.join(backup_dir, f"backup_{timestamp}.zip")
            
            with zipfile.ZipFile(filename, 'w', zipfile.ZIP_DEFLATED) as zipf:
                if os.path.exists("data"):
                    for root, dirs, files in os.walk("data"):
                        for file in files:
                            # 排除截图文件夹，太大了
                            if "snapshots" in root:
                                continue
                            file_path = os.path.join(root, file)
                            arcname = os.path.relpath(file_path, ".")
                            zipf.write(file_path, arcname)
            
            QMessageBox.information(self, "备份成功", f"已创建备份文件:\n{os.path.abspath(filename)}")
        except Exception as e:
            QMessageBox.critical(self, "备份失败", str(e))

    def restore_data(self):
        path, _ = QFileDialog.getOpenFileName(self, "选择备份文件", "", "Zip Files (*.zip)")
        if not path:
            return

        reply = QMessageBox.question(
            self, "确认还原", 
            "还原操作将覆盖当前的配置文件，确定要继续吗？",
            QMessageBox.Yes | QMessageBox.No
        )
        if reply == QMessageBox.Yes:
            try:
                with zipfile.ZipFile(path, 'r') as zipf:
                    zipf.extractall(".")
                QMessageBox.information(self, "还原成功", "数据已还原，请重启程序以生效。")
            except Exception as e:
                QMessageBox.critical(self, "还原失败", str(e))

    def open_folder(self, folder):
        path = os.path.abspath(folder)
        if not os.path.exists(path):
            os.makedirs(path)
        os.startfile(path)
